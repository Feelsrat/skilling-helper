@Composable
private fun SheetScaffold(
    atPeek: Boolean,
    onDismiss: () -> Unit,
    content: @Composable () -> Unit,
) {
    // Skilling Helper build: a desktop-style sidebar docked to the RIGHT edge instead of the bottom sheet. Nothing
    // is drawn or hit-tested over the rest of the screen, so the game stays visible and playable beside the panel,
    // like RuneLite's sidebar on desktop. Close it with the ✕ at its top right or BACK. [atPeek] is unused.
    @Suppress("UNUSED_VARIABLE") val unusedPeek = atPeek
    BoxWithConstraints(Modifier.fillMaxSize()) {
        val density = LocalDensity.current
        val fullW = with(density) { maxWidth.toPx() }
        // Width: dragged by the grip on the left edge and kept; by default the plugin panels' exact pixel width.
        val panelPx = ShUi.clampWidth(ShUi.widthPx.value ?: ShUi.defaultWidthPx(density.density), fullW, density.density)
        val panelW = with(density) { panelPx.toDp() }
        val slide = remember { Animatable(panelPx) }
        LaunchedEffect(panelPx) { slide.animateTo(0f) }
        Surface(
            color = Color(0xFF141414),
            modifier =
                Modifier
                    .align(Alignment.CenterEnd)
                    .width(panelW)
                    .fillMaxHeight()
                    .offset { IntOffset(slide.value.roundToInt(), 0) }
                    // Taps on the panel's empty space must not fall through to the game behind it.
                    .pointerInput(Unit) { detectTapGestures { } },
        ) {
            Box(Modifier.fillMaxSize()) {
                Column(
                    Modifier
                        .fillMaxSize()
                        .windowInsetsPadding(
                            WindowInsets.displayCutout
                                .union(WindowInsets.navigationBars)
                                .union(WindowInsets.statusBars)
                                .only(WindowInsetsSides.End + WindowInsetsSides.Top + WindowInsetsSides.Bottom),
                        ),
                ) {
                    // A slim bar with a small ✕ at the top right closes the sidebar.
                    Box(Modifier.fillMaxWidth().height(26.dp)) {
                        Box(
                            Modifier
                                .align(Alignment.CenterEnd)
                                .size(26.dp)
                                .pointerInput(Unit) { detectTapGestures { onDismiss() } },
                            contentAlignment = Alignment.Center,
                        ) {
                            Text("✕", color = Color(0xFFCCCCCC), fontSize = 15.sp)
                        }
                    }
                    Box(Modifier.weight(1f).fillMaxWidth()) { content() }
                }
                // Drag grip on the left edge: drag to resize the sidebar; the game is resized when you let go.
                Box(
                    Modifier
                        .align(Alignment.CenterStart)
                        .width(12.dp)
                        .fillMaxHeight()
                        .pointerInput(fullW) {
                            detectHorizontalDragGestures(
                                onDragEnd = { ShUi.widthPx.value?.let { ShUi.onWidthCommitted?.invoke(it) } },
                            ) { change, dx ->
                                change.consume()
                                val cur = ShUi.widthPx.value ?: panelPx
                                ShUi.widthPx.value = ShUi.clampWidth(cur - dx, fullW, density.density)
                            }
                        },
                    contentAlignment = Alignment.Center,
                ) {
                    Box(
                        Modifier
                            .size(width = 3.dp, height = 40.dp)
                            .clip(RoundedCornerShape(2.dp))
                            .background(Color(0x55FFFFFF)),
                    )
                }
            }
        }
    }
}
