/** Shared state of the desktop-style icon bar: its width, and which entry the open sidebar shows (-1 = the plugin
 *  launcher, null = closed). */
object ShUi {
    const val STRIP_DP = 40
    val active = mutableStateOf<Int?>(null)

    /** The sidebar width the user dragged to (px), null = default; the host saves it and resizes the game. */
    val widthPx = mutableStateOf<Float?>(null)
    var onWidthCommitted: ((Float) -> Unit)? = null

    /** The plugin panels' exact width on this screen (242 Swing px at the integer sh.panelScale), so they show 1:1. */
    fun defaultWidthPx(density: Float): Float = Math.round(280f * density / 242f).coerceIn(1, 6) * 242f

    fun clampWidth(
        px: Float,
        fullW: Float,
        density: Float,
    ): Float = px.coerceIn(minOf(200f * density, fullW * 0.6f), fullW * 0.6f)
}

/** Desktop-style icon bar, always on the right edge in landscape like RuneLite's: ≡ opens the plugin launcher, and
 *  one icon per plugin panel opens that panel in the sidebar. The host toggles: the active icon closes it again. */
@Composable
fun ShIconStrip(
    onOpen: (Pair<Int, String>?) -> Unit,
    tools: List<ShTool> = emptyList(),
) {
    val conf = androidx.compose.ui.platform.LocalConfiguration.current
    if (conf.screenWidthDp <= conf.screenHeightDp) return
    var panels by remember { mutableStateOf<List<PanelEntry>>(emptyList()) }
    LaunchedEffect(Unit) {
        while (true) {
            val json =
                withContext(Dispatchers.IO) {
                    runCatching { JvmLauncher.onBridge { JvmLauncher.listPanels() }.get() }.getOrNull()
                }
            if (json != null) {
                val p = parsePanels(json)
                if (p.map { it.index to it.tooltip } != panels.map { it.index to it.tooltip }) panels = p
            }
            delay(if (panels.isEmpty()) 2_000 else 10_000)
        }
    }
    val active = ShUi.active.value
    // The touch helpers (keyboard, sticky Shift/Ctrl) latch outside this view, so re-read their state now and then.
    var toolTick by remember { mutableStateOf(0) }
    LaunchedEffect(tools.size) {
        while (tools.isNotEmpty()) {
            delay(400)
            toolTick++
        }
    }
    Column(
        Modifier
            .fillMaxSize()
            .background(Color(0xFF1B1B1B))
            .windowInsetsPadding(
                WindowInsets.displayCutout
                    .union(WindowInsets.navigationBars)
                    .union(WindowInsets.statusBars)
                    .only(WindowInsetsSides.End + WindowInsetsSides.Top + WindowInsetsSides.Bottom),
            ).verticalScroll(rememberScrollState()),
        horizontalAlignment = Alignment.CenterHorizontally,
    ) {
        ShStripButton(active == -1, "Plugins", { onOpen(null) }) {
            Text("≡", color = Color.White, fontSize = 20.sp)
        }
        for (p in panels) {
            ShStripButton(active == p.index, p.tooltip, { onOpen(p.index to p.tooltip) }) {
                val icon = p.icon
                if (icon != null) {
                    Image(icon.asImageBitmap(), p.tooltip, Modifier.size(22.dp))
                } else {
                    Text(p.tooltip.take(2), color = Color.White, fontSize = 11.sp)
                }
            }
        }
        if (tools.isNotEmpty()) {
            Box(Modifier.padding(vertical = 6.dp).size(width = 24.dp, height = 1.dp).background(Color(0xFF4A4A4A)))
        }
        for (t in tools) {
            @Suppress("UNUSED_EXPRESSION")
            toolTick
            ShStripButton(t.isOn(), t.label, { t.onClick(); toolTick++ }) {
                Text(t.glyph, color = Color.White, fontSize = 17.sp)
            }
        }
    }
}

/** A touch helper shown at the bottom of the icon bar (they used to float over the game's chat). */
class ShTool(val glyph: String, val label: String, val isOn: () -> Boolean, val onClick: () -> Unit)

@Composable
private fun ShStripButton(
    selected: Boolean,
    label: String,
    onClick: () -> Unit,
    content: @Composable () -> Unit,
) {
    Box(
        Modifier
            .padding(vertical = 2.dp)
            .size(36.dp)
            .clip(RoundedCornerShape(6.dp))
            .background(if (selected) Color(0xFF3A3A3A) else Color.Transparent)
            .clickable(onClickLabel = label, onClick = onClick),
        contentAlignment = Alignment.Center,
    ) { content() }
}
/** Click pad: a big tap target on the game's edge. Each tap on it is one left-click at the spot last tapped or
 *  hovered in the game, so the player clicks with a relaxed thumb instead of aiming each time. One touch = one click:
 *  sliding or holding on it sends nothing more. Each click gives a short buzz. */
class ShClickStrip(
    context: android.content.Context,
    private val onTap: () -> Unit,
) : android.view.View(context) {
    private val d = context.resources.displayMetrics.density
    private val bg = android.graphics.Paint().apply { color = 0x66000000; isAntiAlias = true }
    private val ring = android.graphics.Paint().apply {
        color = 0x88FFFFFF.toInt()
        style = android.graphics.Paint.Style.STROKE
        strokeWidth = 2f * d
        isAntiAlias = true
    }
    private val rect = android.graphics.RectF()

    @Suppress("DEPRECATION")
    private val vibrator = context.getSystemService(android.content.Context.VIBRATOR_SERVICE) as? android.os.Vibrator

    override fun onDraw(canvas: android.graphics.Canvas) {
        rect.set(0f, 0f, width.toFloat(), height.toFloat())
        canvas.drawRoundRect(rect, 10f * d, 10f * d, bg)
        canvas.drawCircle(width / 2f, height / 2f, minOf(width, height) * 0.3f, ring)
    }

    @android.annotation.SuppressLint("ClickableViewAccessibility")
    override fun onTouchEvent(ev: android.view.MotionEvent): Boolean {
        if (ev.actionMasked == android.view.MotionEvent.ACTION_DOWN) {
            onTap()
            runCatching { vibrator?.vibrate(android.os.VibrationEffect.createOneShot(15, 120)) }
        }
        return true
    }
}
